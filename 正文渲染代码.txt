import re
import json
import html as html_lib
from typing import Any
from datetime import datetime


async def main(args: Args) -> Output:
    params = args.params

    try:
        raw_input = params.get("input", "")
        input_gsmc = clean_text(params.get("gsmc", ""))
        input_zdr = clean_text(params.get("zdr", ""))

        data = parse_json_deep(raw_input)
        if isinstance(data, list):
            data = first_dict(data)

        if not isinstance(data, dict):
            error_html = render_error_page(
                "输入解析失败，未得到有效JSON对象",
                f"原始输入预览：{clean_text(raw_input)[:500]}"
            )
            return {
                "key0": {
                    "gsmc": input_gsmc, "index": "", "ywlx1": "", "ywlx2": "", "ywlx3": "", "ywlx4": "",
                    "riqi": "", "title": "解析失败", "url": "", "html": error_html,
                    "zdr": input_zdr, "timestamp": current_timestamp(), "download_link": ""
                }
            }

        gsmc = clean_text(data.get("gsmc", "")) or input_gsmc
        index = clean_text(data.get("index", ""))
        ywlx1 = clean_text(data.get("ywlx1", ""))
        ywlx2 = clean_text(data.get("ywlx2", ""))
        ywlx3 = clean_text(data.get("ywlx3", ""))
        ywlx4 = clean_text(data.get("ywlx4", ""))

        riqi = clean_text(data.get("riqi", ""))
        title = clean_text(data.get("title", ""))
        url = clean_url(data.get("url", ""))

        zdr = clean_text(data.get("zdr", "")) or input_zdr
        timestamp = clean_text(data.get("timestamp", "")) or current_timestamp()
        download_link = clean_text(data.get("download_link", ""))

        source = clean_text(data.get("source", ""))
        author = clean_text(data.get("author", ""))
        views = clean_text(data.get("views", ""))

        content_html = normalize_html_output(data.get("content_html", ""))
        # 1. 移除正文中与顶部渲染重复的标题和发布时间
        content_html = deduplicate_content_header(content_html, title, riqi)
        # 2. 【新增】清理正文多余的闭合标签，防止破坏外层容器结构
        content_html = fix_html_tag_balance(content_html)

        rendered_html = render_full_html(
            title=title,
            riqi=riqi,
            source=source,
            author=author,
            views=views,
            content_html=content_html
        )

        return {
            "key0": {
                "gsmc": gsmc,
                "index": index,
                "ywlx1": ywlx1, "ywlx2": ywlx2, "ywlx3": ywlx3, "ywlx4": ywlx4,
                "riqi": riqi, "title": title, "url": url,
                "html": rendered_html,
                "zdr": zdr, "timestamp": timestamp, "download_link": download_link
            }
        }

    except Exception as e:
        error_html = render_error_page(
            f"渲染代码执行异常：{str(e)}",
            "请检查输入格式和代码逻辑"
        )
        return {
            "key0": {
                "gsmc": "", "index": "", "ywlx1": "", "ywlx2": "", "ywlx3": "", "ywlx4": "",
                "riqi": "", "title": "渲染异常", "url": "",
                "html": error_html,
                "zdr": "", "timestamp": current_timestamp(), "download_link": ""
            }
        }

# ============ 工具函数 ============

def normalize_text(text: str) -> str:
    """文本归一化：去除空白、标点、全角符号，用于内容一致性比对"""
    if not text:
        return ""
    text = re.sub(r'\s+', '', text)
    text = re.sub(r'[，。、：:；;""''（）()【】\[\]《》\-—·. ]', '', text)
    return text.strip()


def deduplicate_content_header(content_html: str, target_title: str, target_date: str) -> str:
    """
    以顶部渲染的标题、日期为基准，移除正文里重复的头部内容
    渲染内容优先级最高，正文匹配一致的元素自动删除
    """
    if not content_html:
        return ""
    
    html = content_html
    norm_title = normalize_text(target_title)
    norm_date = normalize_text(target_date)

    # 1. 移除正文里与渲染标题重复的 h1~h3 标题标签
    def replace_title(match):
        tag_content = match.group(0)
        inner_text = re.sub(r'<[^>]+>', '', tag_content)
        if normalize_text(inner_text) == norm_title:
            return ""
        return tag_content

    html = re.sub(
        r'<h[1-3]\b[^>]*>.*?</h[1-3]>',
        replace_title,
        html,
        flags=re.IGNORECASE | re.DOTALL
    )

    # 2. 移除正文里包含「发布时间/日期」字样、且内含目标日期的元信息块
    def replace_date_block(match):
        tag_content = match.group(0)
        inner_text = re.sub(r'<[^>]+>', '', tag_content)
        norm_inner = normalize_text(inner_text)
        if norm_date and norm_date in norm_inner:
            return ""
        return tag_content

    html = re.sub(
        r'<(p|div|span)\b[^>]*>.*?(发布时间|发布日期|日期).*?</\1>',
        replace_date_block,
        html,
        flags=re.IGNORECASE | re.DOTALL
    )

        # 3. 移除所有纯日期单独段落（无前缀文字，仅日期内容）
    # 注意：当 norm_date 为空时不执行，否则会误删 <p><img.../></p> 等无文本内容的标签
    if norm_date:
        def replace_pure_date(match):
            tag_content = match.group(0)
            inner_text = re.sub(r'<[^>]+>', '', tag_content).strip()
            if normalize_text(inner_text) == norm_date:
                return ""
            return tag_content

        html = re.sub(
            r'<(p|div|span)\b[^>]*>.*?</\1>',
            replace_pure_date,
            html,
            flags=re.IGNORECASE | re.DOTALL
        )


    # 4. 清理清洗后产生的多余空行与空段落
    html = re.sub(r'<(p|div)\b[^>]*>\s*</\1>', '', html, flags=re.IGNORECASE)
    html = re.sub(r'\n\s*\n\s*\n+', '\n\n', html)

    return html.strip()


def fix_html_tag_balance(html: str) -> str:
    """修复正文HTML中div标签不匹配的问题，移除多余的闭合div
    避免正文里的 </div> 提前闭合外层渲染容器
    """
    if not html:
        return ""
    
    # 统计开标签和闭标签数量
    open_divs = len(re.findall(r'<div\b[^>]*>', html, flags=re.IGNORECASE))
    close_divs = len(re.findall(r'</div>', html, flags=re.IGNORECASE))
    
    # 闭合标签多于开标签时，移除多余的闭合标签
    if close_divs > open_divs:
        extra_count = close_divs - open_divs
        
        def remove_extra(match):
            nonlocal extra_count
            if extra_count > 0:
                extra_count -= 1
                return ""
            return match.group(0)
        
        html = re.sub(r'</div>', remove_extra, html, flags=re.IGNORECASE)
    
    return html.strip()


def first_dict(items: list) -> Any:
    for item in items:
        parsed = parse_json_deep(item)
        if isinstance(parsed, dict):
            return parsed
    return None


def parse_json_deep(value: Any) -> Any:
    if value is None:
        return None
    if isinstance(value, (dict, list)):
        return value
    text = str(value).strip()
    if not text:
        return None
    text = remove_markdown_code_fence(text)
    current = text
    for _ in range(5):
        if isinstance(current, (dict, list)):
            return current
        if not isinstance(current, str):
            return current
        current = current.strip()
        try:
            current = json.loads(current)
            continue
        except Exception:
            pass
        try:
            json_text = extract_json_array_or_object(current)
            if json_text:
                current = json.loads(json_text)
                continue
        except Exception:
            pass
        break
    return current


def remove_markdown_code_fence(text: str) -> str:
    text = text.strip()
    if text.startswith("```"):
        text = re.sub(r"^```[a-zA-Z]*\s*", "", text)
        text = re.sub(r"\s*```$", "", text)
    return text.strip()


def extract_json_array_or_object(text: str) -> str:
    text = text.strip()
    array_start = text.find("[")
    array_end = text.rfind("]")
    object_start = text.find("{")
    object_end = text.rfind("}")
    if array_start != -1 and array_end != -1 and array_end > array_start:
        return text[array_start:array_end + 1]
    if object_start != -1 and object_end != -1 and object_end > object_start:
        return text[object_start:object_end + 1]
    return ""


def clean_text(value: Any) -> str:
    if value is None:
        return ""
    text = str(value)
    text = text.replace("\\r\\n", "\n")
    text = text.replace("\\n", "\n")
    text = text.replace("\\t", " ")
    text = text.replace('\\"', '"')
    text = text.replace("\\'", "'")
    text = text.strip().strip('"').strip("'")
    if text.lower() == "null":
        return ""
    if re.fullmatch(r"[\s\\n\\r]+", text):
        return ""
    text = re.sub(r"[ \t]+", " ", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    return text.strip()


def clean_url(value: Any) -> str:
    if value is None:
        return ""
    url = str(value).strip()
    url = url.replace("\\r\\n", "")
    url = url.replace("\\n", "")
    url = url.replace("\\t", "")
    url = url.replace('\\"', "")
    url = url.replace("\\'", "")
    url = url.replace('"', "")
    url = url.replace("'", "")
    url = url.replace("http: //", "http://")
    url = url.replace("https: //", "https://")
    url = re.sub(r"\s+", "", url)
    return url


def normalize_html_output(value: Any) -> str:
    if value is None:
        return ""
    text = str(value)
    text = text.replace("\\r\\n", "\n")
    text = text.replace("\\n", "\n")
    text = text.replace("\\t", "    ")
    text = text.replace('\\"', '"')
    text = text.replace("\\'", "'")
    text = text.strip()
    if len(text) >= 2 and text[0] == '"' and text[-1] == '"':
        text = text[1:-1]
    return text.strip()


def escape_html(value: Any) -> str:
    return html_lib.escape(clean_text(value), quote=False)


def escape_attr(value: Any) -> str:
    return html_lib.escape(clean_text(value), quote=True)


def current_timestamp() -> str:
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")
